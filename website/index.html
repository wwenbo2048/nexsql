nexSql AI · 自然语言转 SQL
查询最近 7 天注册的用户，按注册时间倒序，只要前 50 条
SELECT id, username, email, created_at
FROM users
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
ORDER BY created_at DESC
LIMIT 50
再加一个条件：状态必须是 active
-- 新增 status = 'active' 条件
SELECT id, username, email, created_at
FROM users
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
  AND status = 'active'
ORDER BY created_at DESC
LIMIT 50
